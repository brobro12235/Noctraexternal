cmake_minimum_required(VERSION 3.24)
project(Noctra VERSION 1.2.0 LANGUAGES CXX)
set(NOCTRA_UPDATE_SEQUENCE 3 CACHE STRING "Monotonically increasing published release sequence")
set(NOCTRA_UPDATE_REPOSITORY "brobro12235/Noctraexternal")
set(NOCTRA_UPDATE_PUBLIC_KEY "")
if(EXISTS "${CMAKE_SOURCE_DIR}/config/update-public-key.txt")
  file(READ "${CMAKE_SOURCE_DIR}/config/update-public-key.txt" NOCTRA_UPDATE_PUBLIC_KEY)
  string(STRIP "${NOCTRA_UPDATE_PUBLIC_KEY}" NOCTRA_UPDATE_PUBLIC_KEY)
endif()
configure_file(src/update_config.hpp.in "${CMAKE_BINARY_DIR}/generated/update_config.hpp" @ONLY)
# Link the MSVC runtime into every project/dependency target so recipients do
# not need a matching Visual C++ Redistributable or developer DLLs.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Noctra requires 64-bit Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git
  GIT_TAG f5befd2d29e66809cd1110a152e375a7f1981f06 GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(imgui)
# JSON parsing is a new dependency; no JSON parser was previously used here.
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git
  GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03)
FetchContent_MakeAvailable(json)
add_library(noctra_avatar STATIC src/avatar.cpp)
target_include_directories(noctra_avatar PUBLIC src)
target_compile_definitions(noctra_avatar PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(noctra_avatar PUBLIC nlohmann_json::nlohmann_json winhttp windowscodecs ole32)
if(MSVC)
  target_compile_options(noctra_avatar PRIVATE /W4 /WX /permissive- /utf-8)
endif()
add_library(imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_compile_definitions(imgui PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(noctra_core STATIC src/core.cpp src/client.cpp src/preview.cpp src/offsets.cpp)
target_include_directories(noctra_core PUBLIC src)
target_compile_definitions(noctra_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(noctra_core PUBLIC bcrypt psapi winhttp nlohmann_json::nlohmann_json)
add_library(noctra_update STATIC src/update.cpp)
target_include_directories(noctra_update PUBLIC src "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(noctra_update PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(noctra_update PUBLIC nlohmann_json::nlohmann_json bcrypt winhttp shell32 user32)
add_executable(noctra_release tools/update_release.cpp)
target_link_libraries(noctra_release PRIVATE noctra_update crypt32)
if(MSVC)
  target_compile_options(noctra_update PRIVATE /W4 /WX /permissive- /utf-8)
  target_compile_options(noctra_release PRIVATE /W4 /WX /permissive- /utf-8)
endif()
add_executable(Noctra WIN32 src/main.cpp src/ui.cpp)
target_link_libraries(Noctra PRIVATE noctra_core noctra_avatar noctra_update imgui d3d11 dxgi dwmapi)
add_executable(noctra_update_tests tests/update_tests.cpp)
target_link_libraries(noctra_update_tests PRIVATE noctra_update)
if(MSVC)
  target_compile_options(noctra_update_tests PRIVATE /W4 /WX /permissive- /utf-8)
endif()
add_test(NAME updates COMMAND noctra_update_tests)
# Signed local fixture used only to verify failed-startup rollback.
add_executable(noctra_update_failure WIN32 tests/update_failure.cpp)
set_target_properties(Noctra PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/dist"
  RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/Debug")
add_custom_command(TARGET Noctra POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${imgui_SOURCE_DIR}/LICENSE.txt" "$<TARGET_FILE_DIR:Noctra>/Dear-ImGui-LICENSE.txt"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/enq/LICENSE.txt"
  "$<TARGET_FILE_DIR:Noctra>/enQ-LICENSE.txt"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${json_SOURCE_DIR}/LICENSE.MIT"
  "$<TARGET_FILE_DIR:Noctra>/JSON-LICENSE.txt")
if(MSVC)
  target_compile_options(noctra_core PRIVATE /W4 /WX /permissive- /utf-8)
  target_compile_options(Noctra PRIVATE /W4 /WX /permissive- /utf-8)
endif()
enable_testing()
add_executable(noctra_tests tests/core_tests.cpp)
target_link_libraries(noctra_tests PRIVATE noctra_core)
if(MSVC)
  target_compile_options(noctra_tests PRIVATE /W4 /WX /permissive- /utf-8)
endif()
add_test(NAME core COMMAND noctra_tests)
add_executable(noctra_ui_tests tests/ui_tests.cpp src/ui.cpp)
target_link_libraries(noctra_ui_tests PRIVATE noctra_core noctra_update imgui)
if(MSVC)
  target_compile_options(noctra_ui_tests PRIVATE /W4 /WX /permissive- /utf-8)
endif()
add_test(NAME menu COMMAND noctra_ui_tests)
add_executable(noctra_avatar_tests tests/avatar_tests.cpp)
target_link_libraries(noctra_avatar_tests PRIVATE noctra_avatar)
if(MSVC)
  target_compile_options(noctra_avatar_tests PRIVATE /W4 /WX /permissive- /utf-8)
endif()
add_test(NAME avatar COMMAND noctra_avatar_tests)

# Explicit allowlist: never include build files, profiles, diagnostics or secrets.
install(TARGETS Noctra RUNTIME DESTINATION .)
install(FILES "${imgui_SOURCE_DIR}/LICENSE.txt" DESTINATION . RENAME Dear-ImGui-LICENSE.txt)
install(FILES "${CMAKE_SOURCE_DIR}/third_party/enq/LICENSE.txt" DESTINATION . RENAME enQ-LICENSE.txt)
install(FILES "${json_SOURCE_DIR}/LICENSE.MIT" DESTINATION . RENAME JSON-LICENSE.txt)
install(FILES "${CMAKE_SOURCE_DIR}/docs/distribution.txt" DESTINATION . RENAME READ-ME.txt)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME Noctra)
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_FILE_NAME "Noctra-${PROJECT_VERSION}-windows-x64")
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_SOURCE_DIR}/releases")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
include(CPack)
